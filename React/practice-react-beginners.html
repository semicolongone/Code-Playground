Q19 Array State — Add Items useState

Create a WishList. Store an array of items in state. Have a text input + "Add" button. When clicked, add the input value to the array using spread: [...items, newItem]. Never mutate state directly!

const [items, setItems] = useState([]);
// ✗ Wrong: items.push(newItem)  ← mutates state directly!
// ✓ Correct: setItems([...items, newItem])  ← spread creates new array

Your component renders here

Solution
const { useState } = React;
function WishList() {
  const [items, setItems] = useState(["React"]);
  const [input, setInput] = useState("");
  const addItem = () => {
    if (!input.trim()) return;
    setItems([...items, input.trim()]);
    setInput("");
  };
  return (
    <div>
      <div style={{display:"flex",gap:"8px",marginBottom:"10px"}}>
        <input className="react-input" value={input} onChange={(e) => setInput(e.target.value)} placeholder="Add item..." />
        <button className="react-btn" onClick={addItem}>Add</button>
      </div>
      <ul className="react-list">
        {items.map((item, i) => <li key={i}>• {item}</li>)}
      </ul>
    </div>
  );
}

Three-Tier Architecture separates an app into three layers. React maps naturally to this:

Q45 Microservice Architecture — Concept Architecture

Microservice-inspired React means breaking the UI into fully independent "mini-apps" — each component owns its own state, data fetching, and logic. They communicate through well-defined interfaces (props/callbacks), not through shared globals.

MONOLITH APPROACH (❌ avoid) App.js └── One giant component — handles users, cart, chat, notifications all together. Impossible to test or reuse. ────────────────────────────────────────────── MICROSERVICE-INSPIRED REACT (✅) App.js ├── <UserService /> → owns auth state + user data │ └── Exposes: user, login(), logout() │ ├── <CartService /> → owns cart state independently │ └── Exposes: items, addItem(), removeItem(), total │ ├── <NotificationService /> → own polling, own state │ └── Exposes: count, messages, markRead() │ └── <ChatService /> → own WebSocket, own messages └── Exposes: messages, send() Each "service" = a custom hook + a small component tree. They know nothing about each other — only talk through App.js.

const { useState } = React;

// ── AUTH SERVICE ──
function useAuth() {
  const [user, setUser] = useState(null);
  return {
    user,
    login:  () => setUser({ name: "Aman", role: "customer" }),
    logout: () => setUser(null),
  };
}
function AuthService({ user, login, logout }) {
  return (
    <div style={{display:"flex",alignItems:"center",gap:"10px",padding:"6px 0"}}>
      {user
        ? <><span style={{color:"#4ade80"}}>👤 {user.name}</span><button className="react-btn secondary" onClick={logout}>Logout</button></>
        : <button className="react-btn" onClick={login}>Login</button>
      }
    </div>
  );
}

// ── CART SERVICE ──
function useCart() {
  const [items, setItems] = useState([]);
  const addItem    = (product) => setItems(prev => {
    const ex = prev.find(i => i.id === product.id);
    return ex ? prev.map(i => i.id===product.id?{...i,qty:i.qty+1}:i) : [...prev,{...product,qty:1}];
  });
  const removeItem = (id)     => setItems(prev => prev.filter(i => i.id !== id));
  const total      = items.reduce((s,i) => s + i.price*i.qty, 0);
  return { items, addItem, removeItem, total };
}
function CartService({ items, removeItem, total }) {
  return (
    <div style={{background:"#0b1a0e",border:"1px solid #4ade8033",borderRadius:"8px",padding:"10px"}}>
      <p style={{color:"#4ade80",fontWeight:"700",marginBottom:"6px"}}>🛒 Cart ({items.length})</p>
      {items.length === 0 ? <p style={{color:"#8892a4",fontSize:"0.85rem"}}>Empty</p> : items.map(i => (
        <div key={i.id} style={{display:"flex",justifyContent:"space-between",padding:"3px 0",color:"#e2e8f0",fontSize:"0.85rem"}}>
          <span>{i.name} ×{i.qty}</span>
          <span>₹{(i.price*i.qty).toLocaleString()} <button onClick={() => removeItem(i.id)} style={{background:"none",border:"none",color:"#ff6b6b",cursor:"pointer"}}>✕</button></span>
        </div>
      ))}
      {items.length > 0 && <p style={{color:"#4ade80",fontWeight:"700",marginTop:"6px"}}>Total: ₹{total.toLocaleString()}</p>}
    </div>
  );
}

// ── CATALOG SERVICE ──
const CATALOG = [
  {id:1,name:"Laptop",price:45000},{id:2,name:"Mouse",price:800},{id:3,name:"Keyboard",price:1500}
];

            Add
          </button>
        </div>
      ))}
    </div>
  );
}

// ── SHELL APP: Wires independent services together ──
function EcommerceApp() {
  const auth = useAuth();
  const cart = useCart();
  return (
    <div style={{display:"flex",flexDirection:"column",gap:"8px"}}>
      <AuthService user={auth.user} login={auth.login} logout={auth.logout} />

        <CatalogService onAddToCart={cart.addItem} isLoggedIn={!!auth.user} />
        <CartService items={cart.items} removeItem={cart.removeItem} total={cart.total} />
      </div>
    </div>
  );
}

const { useState } = React;
const QUESTIONS = [
  { q:"What hook stores state in React?",    opts:["useEffect","useState","useRef","useMemo"],       ans:1 },
  { q:"JSX uses ___ instead of 'class'?",     opts:["cssName","class","className","styleClass"], ans:2 },
  { q:"What stops a form page from reloading?", opts:["e.stop()","e.halt()","e.preventDefault()","return false"], ans:2 },

function QuizApp() {
  const [idx,      setIdx]      = useState(0);
  const [selected, setSelected] = useState(null);
  const [score,    setScore]    = useState(0);
  const [done,     setDone]     = useState(false);
  const q = QUESTIONS[idx];
  const choose = (i) => { if (selected !== null) return; setSelected(i); if (i === q.ans) setScore(s=>s+1); };
  const next   = () => { idx+1 < QUESTIONS.length ? (setIdx(idx+1),setSelected(null)) : setDone(true); };
  const reset  = () => { setIdx(0);setSelected(null);setScore(0);setDone(false); };
  if (done) return (

    <div>
      <p style={{color:"#8892a4",marginBottom:"6px"}}>Q{idx+1}/{QUESTIONS.length}</p>
      <p style={{color:"#e2e8f0",fontWeight:"600",marginBottom:"10px"}}>{q.q}</p>

            if (i === selected && i !== q.ans) { bg="#2a0b0b"; border="1px solid #ff6b6b"; }
          }
          return (
            <div key={i} onClick={() => choose(i)}
              style={{background:bg,border,borderRadius:"6px",padding:"8px 12px",cursor:"pointer",color:"#e2e8f0"}}>
              {opt}
            </div>
          );
        })}
      </div>

    </div>
  );

const { useState } = React;
const CATS = ["All","Food","Travel","Study","Other"];
function ExpenseTracker() {
  const [expenses, setExpenses] = useState([
    {id:1,name:"Lunch",  amount:150, cat:"Food"  },
    {id:2,name:"Bus",    amount:50,  cat:"Travel"},
    {id:3,name:"Books",  amount:800, cat:"Study" },
  ]);
  const [form, setForm] = useState({name:"",amount:"",cat:"Food"});
  const [filter, setFilter] = useState("All");
  const add = (e) => {
    e.preventDefault();
    if (!form.name || !form.amount) return;
    setExpenses([...expenses, { id:Date.now(), name:form.name, amount:Number(form.amount), cat:form.cat }]);
    setForm({name:"",amount:"",cat:"Food"});
  };
  const del     = (id)  => setExpenses(expenses.filter(e => e.id !== id));
  const visible = filter === "All" ? expenses : expenses.filter(e => e.cat === filter);
  const total   = visible.reduce((s,e) => s + e.amount, 0);
  return (
    <div style={{display:"flex",flexDirection:"column",gap:"12px"}}>
      // Stat bar
      <div style={{display:"flex",justifyContent:"space-between",alignItems:"center"}}>
        <span style={{color:"#8892a4"}}>{visible.length} items</span>
        <span style={{color:"#61dafb",fontWeight:"700",fontSize:"1.2rem"}}>₹{total.toLocaleString()}</span>
      </div>
      // Add form
      <form onSubmit={add} style={{display:"grid",gridTemplateColumns:"1fr 80px 100px auto",gap:"6px"}}>
        <input className="react-input" placeholder="Expense name" value={form.name} onChange={e=>setForm({...form,name:e.target.value})} />
        <input className="react-input" type="number" placeholder="₹" value={form.amount} onChange={e=>setForm({...form,amount:e.target.value})} />
        <select className="react-input" value={form.cat} onChange={e=>setForm({...form,cat:e.target.value})}>
          {CATS.filter(c=>c!=="All").map(c=><option key={c}>{c}</option>)}
        </select>
        <button className="react-btn" type="submit">Add</button>
      </form>
      // Filter tabs
      <div style={{display:"flex",gap:"6px",flexWrap:"wrap"}}>
        {CATS.map(c=>(
          <button key={c} className={"react-btn "+(filter===c?"":"secondary")} style={{padding:"4px 10px",fontSize:"0.8rem"}} onClick={()=>setFilter(c)}>{c}</button>
        ))}
      </div>
      // List
      <div>
        {visible.length === 0 && <p style={{color:"#8892a4"}}>No expenses yet.</p>}
        {visible.map(e=>(
          <div key={e.id} style={{display:"flex",justifyContent:"space-between",alignItems:"center",padding:"7px 0",borderBottom:"1px solid #2a2f45"}}>
            <span style={{color:"#e2e8f0"}}>{e.name}</span>
            <div style={{display:"flex",gap:"8px",alignItems:"center"}}>
              <span className="react-tag">{e.cat}</span>
              <span style={{color:"#61dafb",fontWeight:"700"}}>₹{e.amount}</span>
              <button onClick={()=>del(e.id)} style={{background:"none",border:"none",color:"#ff6b6b",cursor:"pointer"}}>✕</button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
